    # v5.23 — Transitions entre écrans : aller, retour, changement de rubrique ; barres fixes ; moins d'animations.
    page.goto(url); page.wait_for_selector("text=/Bonjour|Bonsoir/", timeout=15000)
    page.evaluate("""() => { window.vtLog = []; new MutationObserver(() => { const v = document.documentElement.dataset.vt; if (v) vtLog.push(v); })
      .observe(document.documentElement, { attributes: true, attributeFilter: ['data-vt'] }); }""")
    dernier = lambda: page.evaluate("vtLog[vtLog.length - 1] || ''")
    attendre = lambda: page.wait_for_timeout(700)

    anim = lambda: page.evaluate("getComputedStyle(document.querySelector('.wrap')).animationName")
    page.click(".bottom [data-v=ruches]"); attendre()
    verif(dernier() == "onglet", "Accueil → Ruchers (barre du bas) : fondu de rubrique")
    verif(page.evaluate("document.documentElement.dataset.vt") is None, "le sens est retiré une fois l'entrée jouée")
    page.click(".subtabs [data-v=varroa]"); a = anim(); page.wait_for_timeout(700)
    verif(a == "vt-monte" and page.evaluate("S.route") == "varroa", "sous-onglet Santé : fondu, écran changé tout de suite")
    page.click(".subtabs [data-v=ruches]"); page.wait_for_timeout(700)
    # v5.32 : le volet Menu remplace Gestion ; un choix dans le volet change de rubrique (fondu).
    page.click(".bottom [data-v=menu]"); page.wait_for_selector("#nlVolet[open]")
    page.click("#nlVolet [data-v=fournisseurs]"); attendre()
    verif(dernier() == "onglet" and page.evaluate("S.route") == "fournisseurs", "Ruchers → Fournisseurs par le volet : fondu")
    page.click(".bottom [data-v=dash]"); attendre()
    page.evaluate("ACT.go('agenda')"); a = anim(); page.wait_for_selector(".u-retour"); attendre()
    verif(dernier() == "aller" and a == "vt-droite", "Accueil → Agenda : le contenu arrive de la droite, sans attendre")
    cible = page.get_attribute(".u-retour", "data-v")
    page.click(".u-retour"); a = anim(); attendre()
    verif(dernier() == "retour" and page.evaluate("S.route") == cible, f"« ‹ … » : retour vers {cible}, l'écran revient de la gauche")
    verif(a == "vt-gauche", "retour : le contenu arrive de la gauche")
    page.evaluate("ACT.go('plus')"); attendre(); page.evaluate("ACT.go('fournisseurs')"); attendre(); page.evaluate("ACT.go('plus')"); attendre()
    verif(dernier() == "retour", "revenir à l'écran précédent sans le bouton : reconnu comme un retour")
    page.evaluate("ACT.go('fournisseurs')"); attendre()
    page.keyboard.press("Escape"); attendre()
    verif(page.evaluate("S.route") == "plus" and dernier() == "retour", "Échap sur ordinateur : retour à l'écran précédent")
    page.evaluate("ACT.go('fournisseurs')"); attendre(); page.evaluate("ACT.newvisite()"); page.wait_for_selector("#v_note", timeout=5000); attendre()
    r0 = page.evaluate("S.route"); page.focus("#v_note"); page.keyboard.press("Escape"); attendre()
    verif(page.evaluate("S.route") == r0, "Échap dans une visite en cours : on reste sur la saisie")

    # Fiche ruche : l'alvéole garde son mouvement, l'écran glisse dans le bon sens.
    page.evaluate(f"ACT.go('ruches')"); attendre()
    page.evaluate(f"ACT.ruche('{R1}')"); page.wait_for_selector(".u-gros-hex", timeout=5000); attendre()
    verif(dernier() == "aller" and page.evaluate("S.route") == "ruche", "Ruchers → fiche ruche : aller")
    verif(page.locator(".u-gros-hex").count() == 1, "fiche ruche affichée")
    page.evaluate("ACT.go('ruches')"); attendre()
    verif(dernier() == "retour", "fiche ruche → Ruchers : retour")

    # Une seule entrée par écran : l'ancienne entrée (« rise ») ne repart pas quand le sens est retiré.
    page.evaluate("""() => { window.entrees = []; document.addEventListener('animationstart', (e) => { if (e.target.classList?.contains('wrap')) entrees.push(e.animationName); }, true); }""")
    page.click(".bottom [data-v=dash]"); page.wait_for_timeout(1200)
    verif(page.evaluate("entrees") == ["vt-monte"], "changer d'écran : le contenu n'entre qu'une fois")
    page.evaluate("ACT.go('agenda')"); page.wait_for_timeout(1200)
    verif(page.evaluate("entrees") == ["vt-monte", "vt-droite"], "ouvrir un écran : une seule entrée, de la droite")
    page.evaluate("ACT.go('ruches')"); attendre()

    n = page.evaluate("vtLog.length")
    page.emulate_media(reduced_motion="reduce")
    page.click(".bottom [data-v=dash]"); attendre()
    verif(page.evaluate("vtLog.length") == n and page.evaluate("S.route") == "dash", "moins d'animations : l'écran change d'un coup")
    verif(page.locator(".bottom").count() == 1 and page.locator(".topbar").count() == 1, "une seule barre du haut et du bas (nom de transition unique)")
