    # v5.22 — Météo lisible : verdict « maintenant » et pourquoi, jauge des jours, frise de la journée, détail d'une heure,
    # créneaux du jour avec « Prévoir », tableau complet replié, frise courte sur la carte du rucher.
    # Avec CAPTURES=<dossier>, enregistre l'écran à 390 et 1 280 px.

    cap = os.environ.get("CAPTURES")
    vide = {"acces": "", "proprietaire": "", "proprietaire_tel": "", "capacite": None, "environnement": "", "ressources": "", "point_eau": "", "notes": "", "archive_le": None}
    RV = str(uuid.uuid4())
    DATA["ruchers"] = [{"id": RV, "nom": "Verger", "lat": 47.32, "lng": 5.04, **vide}]
    DATA["ruches"][0].update({"rucher_id": RV, "statut": "active", "sortie": "", "lat": 47.32, "lng": 5.04})

    # Six jours d'heures. Aujourd'hui, il est 11 h au rucher : 10 h est passée.
    # Aujourd'hui : 10 h trop frais, 11 h rapide, 12-13 h bon, 14 h trop de vent, 15 h pluie probable.
    # Demain : pluie. Après-demain : bon de 10 h à 16 h. Puis frais (rapide), puis vent.
    jours = [(datetime.date.today() + datetime.timedelta(days=k)).isoformat() for k in range(6)]
    T, P, V, G, H, C, PR = [], [], [], [], [], [], []
    def heure(k, h):
        if k == 0: return {10: (12, 5, 8), 11: (14, 5, 8), 12: (17, 5, 10), 13: (19, 10, 12), 14: (19, 10, 31), 15: (18, 60, 14)}.get(h, (11 if h < 10 else 16, 10, 8))
        if k == 1: return (13, 85, 15)
        if k == 2: return (21, 5, 9)
        if k == 3: return (14, 5, 9)
        return (18, 5, 36)
    time = []
    for k, j in enumerate(jours):
        for h in range(24):
            t, p, v = heure(k, h); time.append(f"{j}T{h:02d}:00")
            T.append(t); P.append(p); V.append(v); G.append(v + 12); H.append(70 if k == 1 else 55); C.append(61 if p >= 60 else 1 if t >= 17 else 3)
            PR.append(0.6 if k == 1 and 9 <= h <= 17 else 0)
    meteo = {"current": {"time": f"{jours[0]}T11:00", "temperature_2m": 14, "wind_speed_10m": 8, "precipitation": 0, "weather_code": 2},
             "hourly": {"time": time, "temperature_2m": T, "precipitation_probability": P, "precipitation": PR, "wind_speed_10m": V,
                        "wind_gusts_10m": G, "relative_humidity_2m": H, "weather_code": C},
             "daily": {"time": jours, "temperature_2m_max": [19, 13, 21, 14, 18, 18], "weather_code": [2, 61, 1, 3, 3, 3]}}
    ctx.route("https://api.open-meteo.com/**", lambda rt: rt.fulfill(status=200, headers={"Content-Type": "application/json", "Access-Control-Allow-Origin": "*"}, body=json.dumps(meteo)))
    ctx.route("https://tile.openstreetmap.org/**", lambda rt: rt.abort())
