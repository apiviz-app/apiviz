    page.add_init_script("""
      window.__recos = [];
      window.SpeechRecognition = window.webkitSpeechRecognition = class { constructor() { window.__recos.push(this); } start() {} stop() { this.onend && this.onend(); } };
      window.dire = (t) => { const r = window.__recos.at(-1); const res = [[{ transcript: t }]]; res[0].isFinal = true; res.length = 1; r.onresult({ resultIndex: 0, results: res }); };
    """)
    page.set_viewport_size({"width": 390, "height": 844})
    page.goto(url); page.wait_for_selector("text=/Bonjour|Bonsoir/", timeout=15000)

    d = page.evaluate("id => Object.fromEntries(TN.decouper('Ruche 2, reine vue. Ruche 1 : provisions bonnes. Ruche 3 faible activité', id).res)", R[:3])

    # Départ : priorité et trois modes, sans menu du bas
    page.evaluate("ACT.v3tournee('Verger')"); page.wait_for_selector(".tn-e-depart")
    verif(page.evaluate("S.tournee.ids[0]") == R[1], "la ruche à surveiller passe en premier")
    verif("commence par Ruche 2" in page.locator(".tn-say").inner_text(), "phrase de départ")
    verif(page.locator(".tn-mode").count() == 3 and page.locator(".bottom").count() == 0, "trois modes, écran de parcours")
    verif(page.evaluate(plein)["large"], "pas de défilement horizontal")
    shot("tn-depart")

    # Mode libre
    page.click(".tn-mode >> nth=0"); page.wait_for_selector(".tn-e-libre")
    page.evaluate("dire('Ruche 2, reine toujours pas vue, provisions moyennes, 38 kilos, activité normale')")
    page.evaluate("dire('Ruche 1, grosse activité, reine vue, couvain compact, faudra poser une hausse')")
    verif("Ruche 1 en cours" in page.locator(".tn-suivi").inner_text(), "API suit la ruche dictée")
    t = page.locator(".tn-compris").inner_text()
    verif("Reine non vue" in t and "Activité forte" in t and "Provisions ?" in t, "chaque ruche se remplit, ce qui manque se voit")
    shot("tn-libre")
    verif(page.evaluate(plein)["vide"] <= 16, "dictée sans zone vide : %s" % page.evaluate(plein))

    page.click("[data-act=tnmodif] >> nth=1"); page.wait_for_selector(".tn-e-manuel")

    page.click("[data-act=tnenreg]"); page.wait_for_selector(".tn-e-bilan", timeout=10000)
