    # IT-10 (v5.74, migration 034) — tournée dans l'ordre de pose (AP-US-045) : proposée dès qu'un rucher est rangé,

    # Avec CAPTURES=<dossier>, captures à 375 et 1 280 px.
    import datetime, os
    cap = os.environ.get("CAPTURES")
    if cap: os.makedirs(cap, exist_ok=True)
    shot = lambda nom: cap and (page.wait_for_timeout(300), page.screenshot(path=f"{cap}/{nom}.png", full_page=True))
    J = lambda n: (datetime.date.today() - datetime.timedelta(days=n)).isoformat()
    noms = [("Ruche 1", "Verger", "Bonne", 3), ("Ruche 2", "Verger", "À surveiller", 1), ("Ruche 3", "Verger", "Critique", 4), ("Ruche 4", "Verger", "Bonne", 2), ("Ruche 5", "Bois", "Bonne", None)]
    DATA["ruches"] = [{"id": str(uuid.uuid4()), "nom": a, "rucher": b, "etat": c, "ordre_pose": o, "code_qr": "", "type_ruche": "Dadant", "annee_reine": 2025, "notes": "", "hausses": 0} for a, b, c, o in noms]
    R = [r["id"] for r in DATA["ruches"]]
    DATA["visites"] = []
    page.add_init_script("localStorage.setItem('apidoo-tournee-modele', 'standard')")
    rangements = []
    def rpc_ranger(route):
        p = (route.request.post_data_json or {}).get("p", {}); rangements.append(p["ruches"])
        for i, rid in enumerate(p["ruches"]):
            for r in DATA["ruches"]:
                if r["id"] == rid: r["ordre_pose"] = i + 1
        return route.fulfill(status=200, headers={"Access-Control-Allow-Origin": "*", "Content-Type": "application/json"}, body=json.dumps(len(p["ruches"])))
    ctx.route("**/rest/v1/rpc/ranger_ruches", lambda r: r.abort("internetdisconnected") if etat["coupe"] else rpc_ranger(r))
    page.set_viewport_size({"width": 375, "height": 812})
    page.goto(url); page.wait_for_selector("text=/Bonjour|Bonsoir/", timeout=15000)

    noms_tn = lambda: [page.evaluate("byId(S.d.ruches, '%s').nom" % i) for i in ids()]

    page.click(".tn-ordre [data-act=tnordre][data-v=pose]")
